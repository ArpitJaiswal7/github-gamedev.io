// 1. Unity C# JsonUtility Serializable Contract

          

            Flat structs (zero nested Dictionaries) so the exact same JSON schema loads in Unity C# (Application.persistentDataPath) and WebGL / HTML5 (localStorage).
          

          
[Serializable]
public struct RunSubmission {
    public string gameId;   // "meowsweeper" | "calm_jigsaw" | "jigsaw_solitaire"
    public int rawScore;    // TimeMs (Best) or Points (Cumulative)
    public int moves;
    public int gridSize;
}

// Single boundary call at end of a finished run:
LeaderboardController.Instance.SubmitScore(new RunSubmission {
    gameId = "meowsweeper",
    rawScore = 38400, // 38.40s
    moves = 34,
    gridSize = 6
});
